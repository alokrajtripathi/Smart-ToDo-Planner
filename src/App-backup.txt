import { useState, useEffect } from "react";
import {
  LayoutDashboard,
  CalendarDays,
  CheckCircle2,
  Clock,
  Plus,
  Search,
  Bell,
  Settings,
  Menu,
  X,
  Trash2,
  Circle,
  ListTodo,
  Pencil,
  AlertCircle,
} from "lucide-react";

function App() {
  // Load saved tasks
  const [tasks, setTasks] = useState(() => {
    try {
      const savedTasks = localStorage.getItem("smart-todo-tasks");
      return savedTasks ? JSON.parse(savedTasks) : [];
    } catch {
      return [];
    }
  });

  const [filter, setFilter] = useState("All");
  const [search, setSearch] = useState("");
  const [taskInput, setTaskInput] = useState("");
  const [priority, setPriority] = useState("Medium");
  const [dueDate, setDueDate] = useState("");
  const [showModal, setShowModal] = useState(false);
  const [sidebarOpen, setSidebarOpen] = useState(false);
  const [editingId, setEditingId] = useState(null);

  const today = new Date().toLocaleDateString("en-CA");

  // Save tasks automatically
  useEffect(() => {
    localStorage.setItem("smart-todo-tasks", JSON.stringify(tasks));
  }, [tasks]);

  // Open Add Modal
  const openAddModal = () => {
    setEditingId(null);
    setTaskInput("");
    setPriority("Medium");
    setDueDate("");
    setShowModal(true);
  };

  // Open Edit Modal
  const openEditModal = (task) => {
    setEditingId(task.id);
    setTaskInput(task.title);
    setPriority(task.priority);
    setDueDate(task.dueDate || "");
    setShowModal(true);
  };

  // Add or Update Task
  const saveTask = (e) => {
    e.preventDefault();

    if (!taskInput.trim()) return;

    if (editingId !== null) {
      setTasks((prevTasks) =>
        prevTasks.map((task) =>
          task.id === editingId
            ? {
                ...task,
                title: taskInput.trim(),
                priority,
                dueDate,
              }
            : task
        )
      );
    } else {
      const newTask = {
        id: Date.now(),
        title: taskInput.trim(),
        priority,
        completed: false,
        date: today,
        dueDate,
      };

      setTasks((prevTasks) => [...prevTasks, newTask]);
    }

    setShowModal(false);
    setEditingId(null);
    setTaskInput("");
    setPriority("Medium");
    setDueDate("");
  };

  // Complete Task
  const toggleTask = (id) => {
    setTasks((prevTasks) =>
      prevTasks.map((task) =>
        task.id === id
          ? { ...task, completed: !task.completed }
          : task
      )
    );
  };

  // Delete Task
  const deleteTask = (id) => {
    setTasks((prevTasks) =>
      prevTasks.filter((task) => task.id !== id)
    );
  };

  // Statistics
  const completed = tasks.filter(
    (task) => task.completed
  ).length;

  const pending = tasks.length - completed;

  const overdue = tasks.filter(
    (task) =>
      !task.completed &&
      task.dueDate &&
      task.dueDate < today
  ).length;

  const progress = tasks.length
    ? Math.round((completed / tasks.length) * 100)
    : 0;

  // Search and Filter
  const filteredTasks = tasks.filter((task) => {
    const matchesFilter =
      filter === "All" ||
      (filter === "Active" && !task.completed) ||
      (filter === "Completed" && task.completed);

    const matchesSearch = task.title
      .toLowerCase()
      .includes(search.toLowerCase());

    return matchesFilter && matchesSearch;
  });

  // Priority Colors
  const priorityColors = {
    High: "bg-red-100 text-red-600",
    Medium: "bg-amber-100 text-amber-600",
    Low: "bg-green-100 text-green-600",
  };

  return (
    <div className="min-h-screen bg-slate-50 text-slate-800 md:flex">

      {/* SIDEBAR */}
      <aside
        className={`fixed inset-y-0 left-0 z-40 w-64 bg-white border-r border-slate-200 p-6 transition-transform md:static md:translate-x-0 ${
          sidebarOpen ? "translate-x-0" : "-translate-x-full"
        }`}
      >
        <div className="flex items-center gap-3 mb-12">
          <div className="bg-indigo-600 text-white p-2 rounded-xl">
            <CheckCircle2 size={25} />
          </div>

          <h1 className="text-xl font-bold">TaskFlow</h1>

          <button
            className="ml-auto md:hidden"
            onClick={() => setSidebarOpen(false)}
          >
            <X />
          </button>
        </div>

        <p className="text-xs font-semibold text-slate-400 mb-4">
          MENU
        </p>

        <nav className="space-y-2">

          <button
            onClick={() => setFilter("All")}
            className={`flex items-center gap-3 w-full p-3 rounded-xl ${
              filter === "All"
                ? "bg-indigo-50 text-indigo-600 font-semibold"
                : "hover:bg-slate-100"
            }`}
          >
            <LayoutDashboard size={19} />
            Dashboard
          </button>

          <button
            onClick={() => setFilter("Active")}
            className={`flex items-center gap-3 w-full p-3 rounded-xl ${
              filter === "Active"
                ? "bg-indigo-50 text-indigo-600"
                : "hover:bg-slate-100"
            }`}
          >
            <CalendarDays size={19} />
            My Tasks
          </button>

          <button
            onClick={() => setFilter("Completed")}
            className={`flex items-center gap-3 w-full p-3 rounded-xl ${
              filter === "Completed"
                ? "bg-indigo-50 text-indigo-600"
                : "hover:bg-slate-100"
            }`}
          >
            <CheckCircle2 size={19} />
            Completed
          </button>

        </nav>

        <div className="absolute bottom-8 left-6 right-6">
          <button className="flex items-center gap-3 p-3 text-slate-500">
            <Settings size={19} />
            Settings
          </button>
        </div>
      </aside>

      {/* MAIN CONTENT */}
      <main className="flex-1 min-w-0 p-5 md:p-10">

        {/* HEADER */}
        <header className="flex justify-between items-center mb-10">

          <div className="flex items-center gap-3">

            <button
              className="md:hidden"
              onClick={() => setSidebarOpen(true)}
            >
              <Menu />
            </button>

            <div>
              <h2 className="text-2xl font-bold">
                Dashboard
              </h2>

              <p className="text-slate-500 text-sm">
                Manage your daily activities.
              </p>
            </div>

          </div>

          <div className="flex items-center gap-4">
            <Bell className="text-slate-500" />

            <div className="w-10 h-10 rounded-full bg-indigo-100 text-indigo-700 flex items-center justify-center font-bold">
              A
            </div>
          </div>

        </header>

        {/* WELCOME */}
        <section className="bg-indigo-600 rounded-2xl p-7 text-white mb-8">

          <h2 className="text-2xl font-bold mb-2">
            Welcome Back! 👋
          </h2>

          <p className="text-indigo-100">
            Stay organized and make today productive.
          </p>

          <button
            onClick={openAddModal}
            className="mt-5 bg-white text-indigo-600 px-5 py-3 rounded-xl font-semibold flex items-center gap-2 hover:bg-indigo-50"
          >
            <Plus size={19} />
            Add New Task
          </button>

        </section>

        {/* STATISTICS */}
        <section className="grid grid-cols-1 sm:grid-cols-2 lg:grid-cols-4 gap-5 mb-8">

          <div className="bg-white p-6 rounded-2xl border border-slate-100">
            <div className="flex justify-between">
              <p className="text-slate-500">Total Tasks</p>
              <ListTodo className="text-indigo-600" />
            </div>
            <h2 className="text-3xl font-bold mt-3">
              {tasks.length}
            </h2>
          </div>

          <div className="bg-white p-6 rounded-2xl border border-slate-100">
            <div className="flex justify-between">
              <p className="text-slate-500">Completed</p>
              <CheckCircle2 className="text-green-600" />
            </div>
            <h2 className="text-3xl font-bold mt-3">
              {completed}
            </h2>
          </div>

          <div className="bg-white p-6 rounded-2xl border border-slate-100">
            <div className="flex justify-between">
              <p className="text-slate-500">Pending</p>
              <Clock className="text-amber-500" />
            </div>
            <h2 className="text-3xl font-bold mt-3">
              {pending}
            </h2>
          </div>

          <div className="bg-white p-6 rounded-2xl border border-slate-100">
            <div className="flex justify-between">
              <p className="text-slate-500">Overdue</p>
              <AlertCircle className="text-red-500" />
            </div>
            <h2 className="text-3xl font-bold mt-3 text-red-500">
              {overdue}
            </h2>
          </div>

        </section>

        {/* PROGRESS */}
        <section className="bg-white rounded-2xl p-6 mb-8 border border-slate-100">

          <div className="flex justify-between mb-3">
            <h3 className="font-bold">Overall Progress</h3>

            <span className="text-indigo-600 font-bold">
              {progress}%
            </span>
          </div>

          <div className="h-3 bg-slate-100 rounded-full overflow-hidden">
            <div
              className="h-full bg-indigo-600 rounded-full transition-all"
              style={{ width: `${progress}%` }}
            />
          </div>

        </section>

        {/* TASK LIST */}
        <section className="bg-white rounded-2xl p-6 border border-slate-100">

          <div className="flex flex-col sm:flex-row justify-between gap-4 mb-6">

            <h3 className="text-xl font-bold">My Tasks</h3>

            <div className="relative">
              <Search
                size={18}
                className="absolute left-3 top-3 text-slate-400"
              />

              <input
                value={search}
                onChange={(e) => setSearch(e.target.value)}
                placeholder="Search tasks..."
                className="border border-slate-200 rounded-xl pl-10 pr-4 py-2 outline-none focus:border-indigo-500"
              />
            </div>

          </div>

          {/* FILTERS */}
          <div className="flex gap-2 mb-6 flex-wrap">

            {["All", "Active", "Completed"].map((item) => (
              <button
                key={item}
                onClick={() => setFilter(item)}
                className={`px-4 py-2 rounded-lg text-sm ${
                  filter === item
                    ? "bg-indigo-600 text-white"
                    : "bg-slate-100 text-slate-600"
                }`}
              >
                {item}
              </button>
            ))}

          </div>

          {/* TASK ITEMS */}
          <div className="space-y-3">

            {filteredTasks.length === 0 ? (

              <div className="text-center py-12 text-slate-400">
                <ListTodo size={40} className="mx-auto mb-3" />
                <p>No tasks found. Add your first task!</p>
              </div>

            ) : (

              filteredTasks.map((task) => {

                const isOverdue =
                  !task.completed &&
                  task.dueDate &&
                  task.dueDate < today;

                return (
                  <div
                    key={task.id}
                    className={`flex flex-wrap items-center gap-3 p-4 border rounded-xl transition ${
                      isOverdue
                        ? "border-red-300 bg-red-50"
                        : "border-slate-100 hover:bg-slate-50"
                    }`}
                  >

                    <button onClick={() => toggleTask(task.id)}>
                      {task.completed ? (
                        <CheckCircle2 className="text-green-600" />
                      ) : (
                        <Circle className="text-slate-400" />
                      )}
                    </button>

                    <div className="flex-1 min-w-0">

                      <p className={`font-medium break-words ${
                        task.completed
                          ? "line-through text-slate-400"
                          : ""
                      }`}>
                        {task.title}
                      </p>

                      <div className="flex flex-wrap gap-3 mt-2 text-xs">

                        <span className="text-slate-400">
                          Created: {task.date || "Earlier"}
                        </span>

                        {task.dueDate && (
                          <span className={
                            isOverdue
                              ? "text-red-600 font-semibold"
                              : "text-slate-500"
                          }>
                            {isOverdue ? "Overdue: " : "Due: "}
                            {task.dueDate}
                          </span>
                        )}

                      </div>

                      {isOverdue && (
                        <p className="text-xs text-red-600 flex items-center gap-1 mt-2">
                          <AlertCircle size={14} />
                          Deadline has passed!
                        </p>
                      )}

                    </div>

                    <span className={`text-xs px-3 py-1 rounded-full ${
                      priorityColors[task.priority] || priorityColors.Medium
                    }`}>
                      {task.priority || "Medium"}
                    </span>

                    <button
                      onClick={() => openEditModal(task)}
                      className="text-slate-400 hover:text-indigo-600 p-1"
                      title="Edit Task"
                    >
                      <Pencil size={18} />
                    </button>

                    <button
                      onClick={() => deleteTask(task.id)}
                      className="text-slate-400 hover:text-red-500 p-1"
                      title="Delete Task"
                    >
                      <Trash2 size={18} />
                    </button>

                  </div>
                );
              })

            )}

          </div>

        </section>

      </main>

      {/* ADD / EDIT MODAL */}
      {showModal && (

        <div className="fixed inset-0 z-50 bg-black/40 flex items-center justify-center p-4">

          <form
            onSubmit={saveTask}
            className="bg-white rounded-2xl p-7 w-full max-w-md shadow-xl"
          >

            <div className="flex justify-between items-center mb-6">

              <h2 className="text-xl font-bold">
                {editingId !== null ? "Edit Task" : "Create New Task"}
              </h2>

              <button
                type="button"
                onClick={() => setShowModal(false)}
              >
                <X className="text-slate-500" />
              </button>

            </div>

            <label className="text-sm font-medium">
              Task Name
            </label>

            <input
              autoFocus
              required
              value={taskInput}
              onChange={(e) => setTaskInput(e.target.value)}
              placeholder="Enter your task..."
              className="w-full border border-slate-200 rounded-xl p-3 mt-2 mb-5 outline-none focus:border-indigo-500"
            />

            <label className="text-sm font-medium">
              Priority
            </label>

            <select
              value={priority}
              onChange={(e) => setPriority(e.target.value)}
              className="w-full border border-slate-200 rounded-xl p-3 mt-2 mb-5"
            >
              <option>High</option>
              <option>Medium</option>
              <option>Low</option>
            </select>

            <label className="text-sm font-medium">
              Due Date
            </label>

            <input
              type="date"
              min={today}
              value={dueDate}
              onChange={(e) => setDueDate(e.target.value)}
              className="w-full border border-slate-200 rounded-xl p-3 mt-2 mb-6 outline-none focus:border-indigo-500"
            />

            <div className="flex gap-3">

              <button
                type="button"
                onClick={() => setShowModal(false)}
                className="flex-1 border border-slate-200 p-3 rounded-xl"
              >
                Cancel
              </button>

              <button
                type="submit"
                className="flex-1 bg-indigo-600 text-white p-3 rounded-xl hover:bg-indigo-700"
              >
                {editingId !== null ? "Save Changes" : "Add Task"}
              </button>

            </div>

          </form>

        </div>

      )}

    </div>
  );
}

export default App;
